# Génizio — règles d'exploration
# Principe : tout ce qui est vitrine est ouvert, tout ce qui est compte ou
# donnée d'enfant est fermé.
# (régénéré le 2026-08-06)

User-agent: *
Allow: /$
Allow: /guides
Allow: /parrainage
Allow: /terms
Allow: /privacy
Allow: /mentions-legales

# Espaces authentifiés et données personnelles : jamais indexés.
# /rejoindre/* contient les liens d'inscription des campagnes partenaires —
# ils sont destinés à être partagés par l'organisation à ses familles, pas
# trouvés par un moteur de recherche.
# /boutique, /laboratory et /nouveautes sont des pages d'application (hors vitrine) :
# elles restent explorables mais portent un <meta noindex> — voir leurs routes.
Disallow: /admin
Disallow: /organisation
Disallow: /supervisor
Disallow: /profiles
Disallow: /profile
Disallow: /auth
Disallow: /rejoindre
Disallow: /s/
